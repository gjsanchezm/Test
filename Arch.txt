import os
import shutil

for f in ["pubring.kbx", "trustdb.gpg"]:
    path = os.path.join(gnupghome, f)
    if os.path.exists(path):
        os.remove(path)

priv = os.path.join(gnupghome, "private-keys-v1.d")
if os.path.exists(priv):
    shutil.rmtree(priv)
